"현금" : (combined.includes('이체') ? "계좌이체" : "카드"), note: `[은행] ${noteText}` }; if (db) db.ref(`ledger/${id}`).set(item); importedCount++; } }); alert(`엑셀 연동 성공! 총 ${importedCount}건의 거래 내역이 회계 장부에 정상 등록되었습니다.`); } else { const wb = XLSX.read(data, { type: 'binary' }); const ws = wb.Sheets[wb.SheetNames[0]]; const json = XLSX.utils.sheet_to_json(ws, { header: 1 }); let importedCount = 0; json.forEach((row, idx) => { if (idx === 0 || !row || row.length === 0) return; const rowStr = row.join(' '); const dateMatch = rowStr.match(/\d{4}[-/.]\d{1,2}[-/.]\d{1,2}/); const date = dateMatch ? dateMatch[0].replace(/\./g, '-') : new Date().toISOString().split('T')[0]; let withdrawal = Number(String(row[2] || '').replace(/[^0-9]/g, '')) || 0; let deposit = Number(String(row[3] || '').replace(/[^0-9]/g, '')) || 0; if (withdrawal > 0 || deposit > 0) { const id = String(Date.now() + idx + Math.random()); const content = row[5] || ''; const memo = row[6] || ''; const item = { id, date, deposit, withdrawal, method: "카드", note: `[은행] ${[content, memo].filter(Boolean).join(' / ')}` }; if (db) db.ref(`ledger/${id}`).set(item); importedCount++; } }); alert(`엑셀 연동 성공! 총 ${importedCount}건 등록 완료.`); } } catch (err) { console.error(err); alert("엑셀 파일을 처리하는 중 오류가 발생했습니다. 파일 형식을 확인해주세요."); } }; reader.readAsText(file, 'euc-kr'); };
